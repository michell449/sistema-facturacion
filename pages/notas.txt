<?php
// Variables disponibles para el archivo que incluye: $tableRowsHtml, $paginationHtml
$paginationHtml = '';
$tableRowsHtml = '';

require_once __DIR__ . '/class/db.php';
require_once __DIR__ . '/../config.php';

// Variables para paginación: mostrar 25 registros por página
$perPage = 25;
$currentPage = (int)($_GET['p'] ?? 1);
$currentPage = max(1, $currentPage); // Asegura que la página sea al menos 1

// Función de escape de HTML
if (!function_exists('ls_html_escape')) {
function ls_html_escape(string $v): string
{
return htmlspecialchars($v, ENT_QUOTES, 'UTF-8');
}
}

try {
$db = (new Database())->getConnection();

// 1. Contar el total de registros para la paginación
$totalRecordsQuery = "SELECT COUNT(*) FROM facturas";
$stTotal = $db->prepare($totalRecordsQuery);
$stTotal->execute();
$totalRecords = (int)$stTotal->fetchColumn();

// Calcular el total de páginas
$totalPages = (int)ceil($totalRecords / $perPage);
$currentPage = min($currentPage, $totalPages); // Ajustar si la página actual excede el total
$offset = ($currentPage - 1) * $perPage;
$offset = max(0, $offset); // Asegurar que el offset no sea negativo

// 2. Consulta para obtener las facturas de la página actual
$sql = "SELECT
uuid,
serie,
folio,
fecha,
emisor_rfc,
emisor_nombre,
receptor_rfc,
receptor_nombre,
receptor_uso_cfdi,
subtotal,
total,
forma_pago,
metodo_pago,
xml_file,
pdf_file
FROM facturas
ORDER BY fecha DESC
LIMIT :limit OFFSET :offset";

$st = $db->prepare($sql);
// Uso de bindValue con PDO::PARAM_INT para mayor seguridad
$st->bindValue(':limit', $perPage, PDO::PARAM_INT);
$st->bindValue(':offset', $offset, PDO::PARAM_INT);
$st->execute();
$rows = $st->fetchAll(PDO::FETCH_ASSOC);

// 3. Generar las filas de la tabla
if (!$rows) {
$tableRowsHtml = '<tr><td colspan="14" class="text-muted">Sin facturas registradas</td></tr>';
} else {
foreach ($rows as $f) {
$xmlFile = $f['xml_file'] ?? '';
$pdfFile = $f['pdf_file'] ?? '';
// Rutas de archivos (se mantienen igual que en el original)
$xmlPath = '/../../app-m/uploads/xml/' . ls_html_escape($xmlFile);
$pdfPath = '/../../app-m/uploads/pdf/' . ls_html_escape($pdfFile);

$tableRowsHtml .= '<tr>';
$tableRowsHtml .= '<td><small>' . ls_html_escape($f['uuid'] ?? '') . '</small></td>';
$tableRowsHtml .= '<td>' . ls_html_escape($f['serie'] ?? '') . '</td>';
$tableRowsHtml .= '<td>' . ls_html_escape($f['folio'] ?? '') . '</td>';
$tableRowsHtml .= '<td><small>' . ls_html_escape($f['fecha'] ?? '') . '</small></td>';
$tableRowsHtml .= '<td><code>' . ls_html_escape($f['emisor_rfc'] ?? '') . '</code></td>';
$tableRowsHtml .= '<td><small>' . ls_html_escape($f['emisor_nombre'] ?? '') . '</small></td>';
$tableRowsHtml .= '<td><code>' . ls_html_escape($f['receptor_rfc'] ?? '') . '</code></td>';
$tableRowsHtml .= '<td><small>' . ls_html_escape($f['receptor_nombre'] ?? '') . '</small></td>';
$tableRowsHtml .= '<td>' . ls_html_escape($f['receptor_uso_cfdi'] ?? '') . '</td>';
$tableRowsHtml .= '<td>$' . number_format((float)($f['subtotal'] ?? 0), 2) . '</td>';
$tableRowsHtml .= '<td>$' . number_format((float)($f['total'] ?? 0), 2) . '</td>';
$tableRowsHtml .= '<td>' . ls_html_escape($f['forma_pago'] ?? '') . '</td>';
$tableRowsHtml .= '<td>' . ls_html_escape($f['metodo_pago'] ?? '') . '</td>';
$tableRowsHtml .= '<td>
<a href="' . $pdfPath . '" class="text-danger" download title="Descargar PDF"> <i class="fas fa-file-pdf fa-lg">pdf</i></a>
<a href="' . $xmlPath . '" class="text-primary ms-2" download title="Descargar XML"> <i class="fas fa-file-code fa-lg">xml</i></a>
</td>';
$tableRowsHtml .= '</tr>';
}
}

// 4. Generar el HTML de la paginación (utilizando Bootstrap 5)
if ($totalPages > 1) {
$paginationHtml = '<div class="row mt-3"><div class="col-12">';
$paginationHtml .= '<nav aria-label="Paginación de facturas">';
$paginationHtml .= '<ul class="pagination justify-content-center">';

// Botón Anterior
$disabledPrev = ($currentPage <= 1) ? 'disabled' : '';
$prevPage = max(1, $currentPage - 1);
// Nota: La URL se construye asumiendo que 'panel?pg=cargar-facturas' es el punto de entrada.
$paginationHtml .= '<li class="page-item ' . $disabledPrev . '">';
$paginationHtml .= '<a class="page-link" href="panel?pg=cargar-facturas&p=' . $prevPage . '" aria-label="Anterior">Anterior</a>';
$paginationHtml .= '</li>';

// Páginas (se muestra un rango de páginas)
$start = max(1, $currentPage - 2);
$end = min($totalPages, $currentPage + 2);

if ($start > 1) {
$paginationHtml .= '<li class="page-item"><a class="page-link" href="panel?pg=cargar-facturas&p=1">1</a></li>';
if ($start > 2) {
$paginationHtml .= '<li class="page-item disabled"><span class="page-link">...</span></li>';
}
}

for ($i = $start; $i <= $end; $i++) {
$active = ($i === $currentPage) ? 'active' : '';
$paginationHtml .= '<li class="page-item ' . $active . '">';
$paginationHtml .= '<a class="page-link" href="panel?pg=cargar-facturas&p=' . $i . '">' . $i . '</a>';
$paginationHtml .= '</li>';
}

if ($end < $totalPages) {
if ($end < $totalPages - 1) {
$paginationHtml .= '<li class="page-item disabled"><span class="page-link">...</span></li>';
}
$paginationHtml .= '<li class="page-item"><a class="page-link" href="panel?pg=cargar-facturas&p=' . $totalPages . '">' . $totalPages . '</a></li>';
}

// Botón Siguiente
$disabledNext = ($currentPage >= $totalPages) ? 'disabled' : '';
$nextPage = min($totalPages, $currentPage + 1);
$paginationHtml .= '<li class="page-item ' . $disabledNext . '">';
$paginationHtml .= '<a class="page-link" href="panel?pg=cargar-facturas&p=' . $nextPage . '" aria-label="Siguiente">Siguiente</a>';
$paginationHtml .= '</li>';

$paginationHtml .= '</ul>';
$paginationHtml .= '</nav>';
$paginationHtml .= '</div></div>';
}

// La salida de las filas de la tabla se realiza directamente, ya que el archivo es incluido dentro del tbody
echo $tableRowsHtml;

} catch (Throwable $e) {
// En caso de error, se sigue escribiendo la fila en la tabla, pero solo con el mensaje de error.
$tableRowsHtml = '<tr><td colspan="14" class="text-danger">Error: ' . ls_html_escape($e->getMessage()) . '</td></tr>';
echo $tableRowsHtml;
}
