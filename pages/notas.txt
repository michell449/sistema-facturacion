    $downloadDir = __DIR__ . '/../uploads/tmp/';
    $xmlDir = __DIR__ . '/../uploads/tmp/';
    if (!is_dir($downloadDir)) mkdir($downloadDir, 0755, true);

    $service = create_sat_service($fiel);

    try {
        $result = $service->download($input['packageId']);
        $zipFile = $downloadDir . $input['packageId'] . '.zip';
        file_put_contents($zipFile, $result->getPackageContent());
    } catch (\Throwable $e) {
        json_response([
            'success' => false,
            'message' => 'Error al descargar paquete del SAT: ' . $e->getMessage()
        ], 500);
    }

    // Extraer CFDIs del ZIP
    try {
        $cfdiReader = CfdiPackageReader::createFromFile($zipFile);
    } catch (OpenZipFileException $exception) {
        json_response(['success' => false, 'message' => 'Error al abrir el ZIP: ' . $exception->getMessage()], 500);
    }

    $facturas = [];

    foreach ($cfdiReader->cfdis() as $uuid => $content) {
        $tmpFileName = $uuid . '-' . uniqid() . '.xml';
        $tmpFilePath = $xmlDir . $tmpFileName;

        if (!file_put_contents($tmpFilePath, $content)) {
            continue; // si no se pudo guardar el XML, salta
        }

        // Cargar el XML como DOM para extraer datos
        try {
            $xml = new \SimpleXMLElement($content);
            $namespaces = $xml->getNamespaces(true);
            $xml->registerXPathNamespace('cfdi', $namespaces['cfdi']);

            $comprobante = $xml;
            $emisor = $xml->xpath('//cfdi:Emisor')[0] ?? null;
            $receptor = $xml->xpath('//cfdi:Receptor')[0] ?? null;

            $facturas[] = [
                'uuid' => (string) $xml->xpath('//cfdi:Complemento/tfd:TimbreFiscalDigital')[0]['UUID'] ?? '',
                'version' => (string) $comprobante['Version'] ?? '',
                'fecha' => (string) $comprobante['Fecha'] ?? '',
                'subtotal' => (float) $comprobante['SubTotal'] ?? 0.0,
                'total' => (float) $comprobante['Total'] ?? 0.0,
                'moneda' => (string) $comprobante['Moneda'] ?? '',
                'metodo_pago' => (string) $comprobante['MetodoPago'] ?? '',
                'forma_pago' => (string) $comprobante['FormaPago'] ?? '',
                'lugar_expedicion' => (string) $comprobante['LugarExpedicion'] ?? '',
                'no_certificado' => (string) $comprobante['NoCertificado'] ?? '',
                'condiciones_pago' => (string) $comprobante['CondicionesDePago'] ?? '',
                'exportacion' => (string) $comprobante['Exportacion'] ?? '',
                'tipo_comprobante' => (string) $comprobante['TipoDeComprobante'] ?? '',
                'emisor_rfc' => (string) $emisor['Rfc'] ?? '',
                'emisor_nombre' => (string) $emisor['Nombre'] ?? '',
                'emisor_regimen' => (string) $emisor['RegimenFiscal'] ?? '',
                'receptor_rfc' => (string) $receptor['Rfc'] ?? '',
                'receptor_nombre' => (string) $receptor['Nombre'] ?? '',
                'receptor_domicilio' => '', // No siempre disponible
                'receptor_regimen' => (string) $receptor['RegimenFiscalReceptor'] ?? '',
                'receptor_uso_cfdi' => (string) $receptor['UsoCFDI'] ?? '',
                'no_certificado_sat' => (string) $xml->xpath('//cfdi:Complemento/tfd:TimbreFiscalDigital')[0]['NoCertificadoSAT'] ?? '',
                'rfc_prov_certif' => (string) $xml->xpath('//cfdi:Complemento/tfd:TimbreFiscalDigital')[0]['RfcProvCertif'] ?? '',
                '_tmp_file' => basename($tmpFilePath),
                'serie' => (string) $comprobante['Serie'] ?? '',
                'folio' => (string) $comprobante['Folio'] ?? '',
            ];
        } catch (\Throwable $e) {
            // No procesamos si falla el XML
            error_log("Error al procesar XML $uuid: " . $e->getMessage());
            continue;
        }
    }

    // Enviar facturas a guardar-facturas.php
    $response = null;
    try {
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, 'http://localhost/app-m/core/guardar-facturas.php'); // Ajusta según tu ruta real
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode(['items' => $facturas]));
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        $response = curl_exec($ch);
        curl_close($ch);
    } catch (\Throwable $e) {
        json_response(['success' => false, 'message' => 'Error al enviar datos a guardar-facturas.php', 'error' => $e->getMessage()], 500);
    }

    json_response([
        'success' => true,
        'message' => 'Paquete procesado correctamente.',
        'zipFile' => $zipFile,
        'registro_response' => json_decode($response, true)
    ]);
    break;

function respond($data, $code = 200)
{
    http_response_code($code);

    exit;

function createService(Fiel $fiel): Service
{
    $client = new Client(['timeout' => 90]);
    $webClient = new GuzzleWebClient($client);
    return new Service(new FielRequestBuilder($fiel), $webClient);
}

function getFielFromSource(string $rfc): ?Fiel
{
    if (
        !isset($_SESSION['fiel_data']) ||
        !isset($_SESSION['fiel_data'][$rfc]) ||
        empty($_SESSION['fiel_data'][$rfc]['cer_content']) ||
        empty($_SESSION['fiel_data'][$rfc]['key_content']) ||
        empty($_SESSION['fiel_data'][$rfc]['passphrase'])
    ) {
        return null;
    }
    try {
        $fielData = $_SESSION['fiel_data'][$rfc];
        $fiel = Fiel::create(
            $fielData['cer_content'],
            $fielData['key_content'],
            $fielData['passphrase']
        );
        return $fiel->isValid() ? $fiel : null;
    } catch (Throwable $e) {
        error_log("Error al crear la FIEL desde la sesión para $rfc: " . $e->getMessage());
        return null;
    }
}

        $stmt = $db->prepare('SELECT * FROM cf_solicitudes WHERE id_solicitud = ?');
        $stmt->execute([$idSolicitud]);
        $solicitud = $stmt->fetch(PDO::FETCH_ASSOC);
        if ($solicitud) {
            $solicitudes[] = $solicitud;
        }
    } else {
        $stmt = $db->query("
            SELECT * FROM cf_solicitudes
            WHERE estado NOT IN ('terminada', 'error', 'rechazada', 'vencida', 'cancelada')
        ");
        $solicitudes = $stmt->fetchAll(PDO::FETCH_ASSOC);
    }

    if (empty($solicitudes)) {
        respond(['success' => true, 'message' => 'No hay solicitudes que necesiten verificación.', 'verificadas' => 0]);
    }

    $verificadas = 0;
    $errores = [];
    $ahora = date('Y-m-d H:i:s');
    $nuevosEstados = [];

    foreach ($solicitudes as $solicitud) {
        $idLocal = $solicitud['id_solicitud'];
        $requestId = $solicitud['solicitud_id_sat'];
        $rfc = $solicitud['rfc_emisor'] ?: $solicitud['rfc_receptor'];

        if (empty($rfc)) {
            $errores[] = "Error en Solicitud #$idLocal: No tiene un RFC asociado.";
            continue;
        }

        $fiel = getFielFromSource($rfc);
        if (!$fiel) {
            $errores[] = "Error en Solicitud #$idLocal: No se pudo cargar la FIEL para el RFC $rfc. Por favor, autentíquese de nuevo.";
            continue;
        }

        $service = createService($fiel);

        try {
            $verify = $service->verify($requestId);
            $statusRequest = $verify->getStatusRequest();
            $estadoSAT = 'desconocido';

            if ($statusRequest->isFinished()) {
                $estadoSAT = 'terminada';
            } elseif ($statusRequest->isInProgress() || $statusRequest->isAccepted()) {
                $estadoSAT = 'aceptada';
            } elseif ($statusRequest->isRejected()) {
                $estadoSAT = 'rechazada';
            } elseif ($statusRequest->isFailure()) {
                $estadoSAT = 'error';
            } elseif ($statusRequest->isExpired()) {
                $estadoSAT = 'vencida';
            }

            $paquetesList = [];
            if ($estadoSAT === 'terminada') {
                $packageIds = $verify->getPackagesIds();
                foreach ($packageIds as $pid) {
                    $paquetesList[$pid] = [ 
                        'package_id' => $pid,
                        'estado' => 'pendiente',
                        'zip_path' => null,
                        'fecha_descarga' => null,
                        'mensaje_error' => null,
                        'procesado' => 0
                    ];

                $baseTmp = __DIR__ . '/../uploads/tmp';
                @mkdir($baseTmp, 0775, true);

                $rfcSelStmt = $db->prepare('SELECT rfc_emisor, rfc_receptor FROM cf_solicitudes WHERE id_solicitud = ?');
                $rfcSelStmt->execute([$idLocal]);
                $rr = $rfcSelStmt->fetch(PDO::FETCH_ASSOC);
                $rfcForPath = $rr['rfc_emisor'] ?: $rr['rfc_receptor'];

                foreach ($paquetesList as $pid => &$p) {
                    try {

                        // 1. La función download
                        $downloadResult = $service->download($pid);

                        // 2. Obtenemos el contenido del paquete, que está en base64.
                        $packageContent = $downloadResult->getPackageContent();

                        // 3. Decodificamos el contenido de base64 a binario.
                        $zipData = base64_decode($packageContent);

                        if ($zipData === false) {
                            throw new \Exception("El contenido del paquete {$pid} no es un base64 válido.");
                        }

                        $pathDir = $baseTmp . '/' . $rfcForPath . '/' . $idLocal;
                        @mkdir($pathDir, 0775, true);

                        $filename = $pathDir . '/' . $pid . '.zip';

                        // 4. Guardamos el contenido ya decodificado.
                        $bytes = file_put_contents($filename, $zipData);


                        if ($bytes === false) {
                            throw new \Exception("No se pudo escribir el archivo ZIP: $filename");
                        }

                        $relPath = "uploads/tmp/{$rfcForPath}/{$idLocal}/{$pid}.zip";

                        $p['estado'] = 'descargado';
                        $p['zip_path'] = $relPath;
                        $p['fecha_descarga'] = date('Y-m-d H:i:s');
                    } catch (Throwable $e) {
                        $p['estado'] = 'error';
                        $p['mensaje_error'] = $e->getMessage();
                    }
                }
                unset($p);

            $paquetesPrevios = json_decode($solicitud['paquetes_json'] ?? '[]', true);
