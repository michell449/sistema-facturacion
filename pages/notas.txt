<?php
// core/solicitar-descarga.php
require_once __DIR__ . '/../vendor/autoload.php';
require_once __DIR__ . '/../config.php';
require_once __DIR__ . '/class/db.php';

use PhpCfdi\SatWsDescargaMasiva\RequestBuilder\RequestBuilder;
use PhpCfdi\SatWsDescargaMasiva\Service;
use PhpCfdi\SatWsDescargaMasiva\RequestBuilder\FielRequestBuilder\Fiel;
use PhpCfdi\SatWsDescargaMasiva\WebClientFactory;

// --- CONFIGURAR DATOS ---
$rfc = $_POST['rfc'] ?? '';
$fechaInicio = $_POST['fecha_inicio'] ?? '';
$fechaFin = $_POST['fecha_fin'] ?? '';
$tipoSolicitud = $_POST['tipo'] ?? 'CFDI';

// Validar datos básicos
if (!$rfc || !$fechaInicio || !$fechaFin) {
    echo json_encode(['success' => false, 'message' => 'Datos incompletos']);
    exit;
}

try {
    // Cargar FIEL
    $cerFile = __DIR__ . "/../fiel/{$rfc}.cer";
    $keyFile = __DIR__ . "/../fiel/{$rfc}.key";
    $password = '12345678a'; // <-- cámbiala según tu FIEL

    $fiel = new Fiel($cerFile, $keyFile, $password);

    // Crear cliente y servicio
    $webClient = WebClientFactory::create();
    $service = new Service($webClient, $fiel);

    // Crear solicitud
    $builder = new RequestBuilder();
    $request = $builder->issued($fechaInicio, $fechaFin);

    $response = $service->request($request);

    if ($response->getStatus()->isAccepted()) {
        $idSolicitud = $response->getRequestId();

        // --- Registrar en base de datos ---
        $stmt = $conn->prepare("INSERT INTO cf_solicitudes (rfc_emisor, fecha_inicio, fecha_fin, id_solicitud_sat, estado) VALUES (?, ?, ?, ?, 'pendiente')");
        $stmt->bind_param("ssss", $rfc, $fechaInicio, $fechaFin, $idSolicitud);
        $stmt->execute();

        echo json_encode([
            'success' => true,
            'message' => 'Solicitud enviada correctamente.',
            'id_solicitud_sat' => $idSolicitud
        ]);
    } else {
        echo json_encode([
            'success' => false,
            'message' => 'El SAT no aceptó la solicitud: ' . $response->getStatus()->getMessage(),
        ]);
    }
} catch (Throwable $e) {
    echo json_encode(['success' => false, 'message' => $e->getMessage()]);
}
